```python
from __future__ import annotations

import csv
import html
import shutil
import tempfile
import zipfile
from pathlib import Path


ROOT = Path(__file__).resolve().parent
RESULTS_DIR = ROOT / "Trade_DomExport_Results"
OUTPUT = ROOT / "BC_AF_dashboard_export_check_all_tabs.xlsx"


SHEETS = [
    ("Dashboard Summary", None),
    ("Year Summary", "dashboard_check_year_summary_2016_2025_stata.csv"),
    ("Sector 2025 vs 2024", "dashboard_check_sector_2025_vs_2024_stata.csv"),
    ("Subsector 2025 vs 2024", "dashboard_check_subsector_2025_vs_2024_stata.csv"),
    ("Commodity Group", "dashboard_check_commodity_group_2025_vs_2024_stata.csv"),
    ("Commodity Detail", "dashboard_check_commodity_detail_2025_vs_2024_stata.csv"),
    ("Country 2025 vs 2024", "dashboard_check_country_2025_vs_2024_stata.csv"),
    ("Monthly BC 2025", "dashboard_check_monthly_bc_2025_stata.csv"),
    ("US Destinations Top 10", "dashboard_check_us_destinations_top10_2025.csv"),
    ("US Products Top 10", "dashboard_check_us_products_top10_2025.csv"),
]


def read_csv(name: str) -> list[list[str]]:
    with (ROOT / name).open(newline="", encoding="utf-8-sig") as f:
        return [row for row in csv.reader(f)]


def money_bn(value: str) -> str:
    return f"${float(value) / 1_000_000_000:.2f}bn"


def pct(value: str) -> str:
    return f"{float(value) * 100:.2f}%"


def summary_rows() -> list[list[str]]:
    years = read_csv("dashboard_check_year_summary_2016_2025_stata.csv")
    y_header = years[0]
    year_rows = {row[y_header.index("year")]: row for row in years[1:]}
    row_2025 = year_rows["2025"]

    sectors = read_csv("dashboard_check_sector_2025_vs_2024_stata.csv")
    s_header = sectors[0]
    sector_rows = {row[s_header.index("category")]: row for row in sectors[1:]}

    us_destinations = read_csv("dashboard_check_us_destinations_top10_2025.csv")
    us_products = read_csv("dashboard_check_us_products_top10_2025.csv")

    return [
        ["Metric", "Raw value", "Dashboard display", "Source"],
        [
            "BC agrifood exports total, 2025",
            row_2025[y_header.index("bc_value")],
            money_bn(row_2025[y_header.index("bc_value")]),
            "Year Summary",
        ],
        [
            "BC exports YoY change, 2025 vs 2024",
            row_2025[y_header.index("bc_yoy_change_pct")],
            pct(row_2025[y_header.index("bc_yoy_change_pct")]),
            "Year Summary",
        ],
        [
            "Agriculture (excl. aquaculture), 2025",
            sector_rows["Agriculture (excl. aquaculture)"][s_header.index("value_2025")],
            money_bn(sector_rows["Agriculture (excl. aquaculture)"][s_header.index("value_2025")]),
            "Sector 2025 vs 2024",
        ],
        [
            "Seafood & fisheries, 2025",
            sector_rows["Seafood & fisheries"][s_header.index("value_2025")],
            money_bn(sector_rows["Seafood & fisheries"][s_header.index("value_2025")]),
            "Sector 2025 vs 2024",
        ],
        [
            "Secondary manufacturing (excl. seafood), 2025",
            sector_rows["Secondary manufacturing (excl. seafood)"][s_header.index("value_2025")],
            money_bn(sector_rows["Secondary manufacturing (excl. seafood)"][s_header.index("value_2025")]),
            "Sector 2025 vs 2024",
        ],
        [
            "Top US destination, 2025",
            us_destinations[1][1],
            us_destinations[1][4],
            "US Destinations Top 10",
        ],
        [
            "Top US product, 2025",
            us_products[1][1],
            us_products[1][4],
            "US Products Top 10",
        ],
    ]


def cell_xml(value: str) -> str:
    escaped = html.escape("" if value is None else str(value))
    return f'<c t="inlineStr"><is><t>{escaped}</t></is></c>'


def worksheet_xml(rows: list[list[str]]) -> str:
    body = []
    for row in rows:
        body.append("<row>" + "".join(cell_xml(value) for value in row) + "</row>")
    return (
        '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>'
        '<worksheet xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main">'
        "<sheetData>"
        + "".join(body)
        + "</sheetData></worksheet>"
    )


def build_workbook() -> None:
    sheet_rows = []
    for title, csv_name in SHEETS:
        rows = summary_rows() if csv_name is None else read_csv(csv_name)
        sheet_rows.append((title, rows))

    with tempfile.TemporaryDirectory() as tmp_name:
        tmp = Path(tmp_name)
        (tmp / "_rels").mkdir()
        (tmp / "xl" / "_rels").mkdir(parents=True)
        (tmp / "xl" / "worksheets").mkdir(parents=True)

        for idx, (_, rows) in enumerate(sheet_rows, start=1):
            (tmp / "xl" / "worksheets" / f"sheet{idx}.xml").write_text(
                worksheet_xml(rows), encoding="utf-8"
            )

        content_types = [
            '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>',
            '<Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types">',
            '<Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/>',
            '<Default Extension="xml" ContentType="application/xml"/>',
            '<Override PartName="/xl/workbook.xml" ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet.main+xml"/>',
        ]
        for idx in range(1, len(sheet_rows) + 1):
            content_types.append(
                f'<Override PartName="/xl/worksheets/sheet{idx}.xml" '
                'ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.worksheet+xml"/>'
            )
        content_types.append("</Types>")
        (tmp / "[Content_Types].xml").write_text("".join(content_types), encoding="utf-8")

        (tmp / "_rels" / ".rels").write_text(
            '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>'
            '<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">'
            '<Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument" Target="xl/workbook.xml"/>'
            "</Relationships>",
            encoding="utf-8",
        )

        workbook = [
            '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>',
            '<workbook xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main" '
            'xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships"><sheets>',
        ]
        rels = [
            '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>',
            '<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">',
        ]
        for idx, (title, _) in enumerate(sheet_rows, start=1):
            workbook.append(
                f'<sheet name="{html.escape(title)}" sheetId="{idx}" r:id="rId{idx}"/>'
            )
            rels.append(
                f'<Relationship Id="rId{idx}" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/worksheet" Target="worksheets/sheet{idx}.xml"/>'
            )
        workbook.append("</sheets></workbook>")
        rels.append("</Relationships>")
        (tmp / "xl" / "workbook.xml").write_text("".join(workbook), encoding="utf-8")
        (tmp / "xl" / "_rels" / "workbook.xml.rels").write_text(
            "".join(rels), encoding="utf-8"
        )

        if OUTPUT.exists():
            OUTPUT.unlink()
        with zipfile.ZipFile(OUTPUT, "w", zipfile.ZIP_DEFLATED) as zf:
            for path in tmp.rglob("*"):
                if path.is_file():
                    zf.write(path, path.relative_to(tmp).as_posix())

    RESULTS_DIR.mkdir(exist_ok=True)
    shutil.copy2(OUTPUT, RESULTS_DIR / OUTPUT.name)
    shutil.copy2(Path(__file__), RESULTS_DIR / Path(__file__).name)


if __name__ == "__main__":
    build_workbook()
    print(OUTPUT)
```
